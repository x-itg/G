### API使用示例
```javascript
const { CacheMonitor } = require('./cacheMonitor');

// 创建监控实例
const monitor = new CacheMonitor({
    samplingInterval: 5000,
    hitRateThreshold: 0.7,
    memoryUsageThreshold: 0.8
});

// 注册缓存
monitor.registerCache('myCache', cacheInstance, {
    category: 'analysis',
    priority: 'high'
});

// 记录访问
monitor.recordAccess('myCache', 'key-1', true, 25, 'analysis');
```
